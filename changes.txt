+function M.remove_current_block()
+  local bufnr = vim.api.nvim_get_current_buf()
+  local cursor = vim.api.nvim_win_get_cursor(0)
+  local line = cursor[1]
+  local lines = vim.api.nvim_buf_get_lines(bufnr, 0, -1, false)
+
+  -- Look forward and backward to find enclosing [ANSWER:id] or [Q:id]
+  local block_start, block_end
+
+  for i = line, 1, -1 do
+    if lines[i]:match("%[ANSWER:q_%d+%]") or lines[i]:match("%[Q:q_%d+%]") or lines[i]:match("%[C:q_%d+%]") or lines[i]:match("%[PENDING:q_%d+%]") then
+      block_start = i
+      break
+    end
+  end
+
+  if not block_start then
+    vim.notify("[ai-editutor] No question block found under cursor", vim.log.levels.INFO)
+    return
+  end
+
+  for i = line, #lines do
+    if lines[i]:match("%[/ANSWER:q_%d+%]") or lines[i]:match("%[/CODE:q_%d+%]") or (i > line and (lines[i]:match("%[Q:q_%d+%]") or lines[i]:match("%[C:q_%d+%]") or lines[i]:match("%[PENDING:q_%d+%]"))) then
+      block_end = i
+      break
+    end
+    if i == #lines then
+      block_end = i
+    end
+  end
+
+  if not block_end then
+    return
+  end
+
+  -- Remove block lines (keep empty line before if it would leave a gap)
+  local start_remove = block_start - 1 >= 1 and lines[block_start - 1] == "" and block_start - 1 or block_start
